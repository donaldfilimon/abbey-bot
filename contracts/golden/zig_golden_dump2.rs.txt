//! SCRATCH-ONLY golden dumper, batch 2 (grounding, recall, casing, catalog
//! availability, episode writes). Never part of the oracle checkout.
use serde_json::{Value, json};

fn out_dir() -> std::path::PathBuf {
    std::path::PathBuf::from(std::env::var("ABBEY_GOLDEN_DIR").expect("ABBEY_GOLDEN_DIR"))
}
fn write(name: &str, value: &Value) {
    let mut bytes = serde_json::to_vec_pretty(value).unwrap();
    bytes.push(b'\n');
    std::fs::write(out_dir().join(name), bytes).unwrap();
}
fn kind(k: crate::grounding::SpecificKind) -> &'static str {
    use crate::grounding::SpecificKind::*;
    match k { Version => "version", Date => "date", Year => "year", Percentage => "percentage", Statistic => "statistic", Quotation => "quotation" }
}

fn dump_goldens_2() {
    // Casing sets derived from str::to_lowercase's final-sigma rule.
    let mut ci = Vec::new();
    let mut cased = Vec::new();
    for cp in 0..=0x10FFFFu32 {
        let Some(c) = char::from_u32(cp) else { continue };
        let x1 = format!("{c}\u{3A3}").to_lowercase().ends_with('\u{3C2}');
        let x2 = format!("A{c}\u{3A3}").to_lowercase().ends_with('\u{3C2}');
        if x2 && !x1 { ci.push(cp); }
        if x1 { cased.push(cp); }
    }
    fn to_ranges(v: &[u32]) -> Vec<[u32; 2]> {
        let mut out: Vec<[u32; 2]> = Vec::new();
        for &cp in v {
            if let Some(last) = out.last_mut() { if last[1] + 1 == cp || (last[1] + 1 == 0xD800 && cp == 0xE000) { last[1] = cp; continue; } }
            out.push([cp, cp]);
        }
        out
    }
    let mut sigma_cases = Vec::new();
    for s in ["\u{3A3}", "A\u{3A3}", "A\u{3A3}B", "A.\u{3A3}", "A\u{3A3}.", "A\u{3A3}.b", "1\u{3A3}", "\u{3A3}\u{39F}\u{3A3}", "\u{661}\u{3A3}", "A\u{300}\u{3A3}", "A\u{3A3}\u{300}", "A\u{3A3}\u{300}b", "\u{4E00}\u{3A3}", "\u{2160}\u{3A3}"] {
        sigma_cases.push(json!({"input": s, "lower": s.to_lowercase()}));
    }

    // Grounding.
    let inputs: Vec<String> = serde_json::from_str(&std::fs::read_to_string(std::env::var("ABBEY_GROUNDING_INPUTS").unwrap()).unwrap()).unwrap();
    let specifics = |v: &[crate::grounding::Specific]| -> Value {
        Value::Array(v.iter().map(|s| json!([kind(s.kind), s.text, s.key])).collect())
    };
    let mut rows = Vec::new();
    let all_sources: Vec<&str> = inputs.iter().map(String::as_str).filter(|s| std::panic::catch_unwind(|| { let _ = crate::grounding::Grounding::from_sources([*s]); }).is_ok()).collect();
    for (i, reply) in inputs.iter().enumerate() {
        let mut modes: Vec<(String, Vec<&str>)> = vec![("empty".into(), vec![])];
        let mut neighbours = Vec::new();
        if i > 0 { neighbours.push(inputs[i - 1].as_str()); }
        if i + 1 < inputs.len() { neighbours.push(inputs[i + 1].as_str()); }
        modes.push(("neighbours".into(), neighbours));
        modes.push(("self".into(), vec![reply.as_str()]));
        modes.push(("all".into(), all_sources.clone()));
        for (mode, sources) in modes {
            let r = std::panic::catch_unwind(|| {
                let g = crate::grounding::Grounding::from_sources(sources.iter().copied());
                let v = crate::grounding::check(reply, &g);
                (specifics(&v.examined), specifics(&v.ungrounded), v.grounding_empty, crate::grounding::hedged(reply, &v))
            });
            match r {
                Ok((examined, ungrounded, empty, hedged)) => rows.push(json!({"reply": reply, "mode": mode, "sources": sources, "examined": examined, "ungrounded": ungrounded, "grounding_empty": empty, "hedged": hedged})),
                Err(_) => rows.push(json!({"reply": reply, "mode": mode, "sources": sources, "panic": true})),
            }
        }
    }

    // Recall selection.
    let fact_sets: Vec<Vec<String>> = vec![
        vec![],
        vec!["likes rust".into()],
        vec!["likes rust".into(), "has a cat".into(), "writes rust at night about a cat".into()],
        (0..20).map(|i| format!("fact number {i} about topic{}", i % 4)).collect(),
        (0..6).map(|i| format!("{}{}", "long ".repeat(50 + i * 10), i)).collect(),
        vec!["Café crème".into(), "CAFÉ noir".into(), "ΣΟΦΙΑ loves Greek".into(), "tea".into()],
    ];
    let queries = ["", "rust", "a rust question", "topic1 fact", "café", "σοφια greek tea", "number 7", "long 3"];
    let mut recall = Vec::new();
    for (si, facts) in fact_sets.iter().enumerate() {
        for q in queries {
            for (max, budget) in [(8usize, 1200usize), (3, 1200), (8, 200), (0, 1200)] {
                let sel = crate::recall::select(facts, q, max, budget);
                let idx: Vec<usize> = sel.facts.iter().map(|f| facts.iter().position(|x| x.as_str() == *f).unwrap()).collect();
                recall.push(json!({"set": si, "query": q, "max": max, "budget": budget, "chosen": idx, "omitted": sel.omitted}));
            }
        }
    }
    write("grounding.json", &json!({
        "case_ignorable": to_ranges(&ci), "cased": to_ranges(&cased), "sigma": sigma_cases,
        "rows": rows, "recall_sets": fact_sets, "recall": recall,
    }));

    // Catalog availability matrix.

                    inputs_cat.push((format!("{context:?}/p{perm_mode}/r{ready}/f{flags}"), input));

    let mut matrix = Vec::new();
